<script>
  const HOST = "f0b66236ba314325bcc00d6816612a7d.s1.eu.hivemq.cloud";
  const USER = "bricevelour"; 
  const PASS = "";     
  const PORT = 8884;

  const TOPIC_TELEMETRY = "bricevelour/meteo/telemetry";
  const TOPIC_CMD       = "bricevelour/meteo/cmd";
  const TOPIC_STATUS    = "bricevelour/meteo/status";

  const url = `wss://${HOST}:${PORT}/mqtt`;

  const statusEl = document.getElementById("status");
  const statusText = document.getElementById("statusText");
  const statusSub  = document.getElementById("statusSub");

  const elTemp = document.getElementById("temp");
  const elHum  = document.getElementById("hum");
  const elLux  = document.getElementById("lux");
  const elMode = document.getElementById("modeBadge");
  const elAlive= document.getElementById("alive");
  const elLast = document.getElementById("lastSeen");
  const logEl  = document.getElementById("log");

  const btnBeep = document.getElementById("btnBeep");
  const btnTheme = document.getElementById("btnTheme");
  const btnThemeOff = document.getElementById("btnThemeOff");
  const btnClear = document.getElementById("btnClear");
  const toggleBuzzer = document.getElementById("toggleBuzzer");

  function logLine(line){
    const now = new Date();
    const ts = now.toLocaleTimeString("fr-CA");
    logEl.textContent = `[${ts}] ${line}\n` + logEl.textContent;
  }

  function setStatus(ok, text, sub){
    statusEl.classList.toggle("ok", !!ok);
    statusText.textContent = text;
    statusSub.textContent = sub || "—";
  }

  const canvas = document.getElementById("chart");
  const ctx = canvas.getContext("2d");

  const MAX_POINTS = 60;
  const seriesTemp = [];
  const seriesHum  = [];

  function pushPoint(arr, v){
    arr.push(v);
    if (arr.length > MAX_POINTS) arr.shift();
  }

  function minMax(arr){
    let min = Infinity, max = -Infinity;
    for (const v of arr){
      if (typeof v !== "number" || Number.isNaN(v)) continue;
      if (v < min) min = v;
      if (v > max) max = v;
    }
    if (min === Infinity) { min = 0; max = 1; }
    if (min === max) { max = min + 1; }
    return {min, max};
  }

  function drawChart(){
    ctx.clearRect(0,0,canvas.width,canvas.height);

    ctx.globalAlpha = 1;
    ctx.lineWidth = 1;

    ctx.strokeStyle = "rgba(241,239,231,.10)";
    for (let i=1; i<=4; i++){
      const y = (canvas.height/5)*i;
      ctx.beginPath();
      ctx.moveTo(0,y);
      ctx.lineTo(canvas.width,y);
      ctx.stroke();
    }

    const mmT = minMax(seriesTemp);
    const mmH = minMax(seriesHum);

    function mapY(v, min, max){
      const t = (v - min) / (max - min);
      const pad = 22;
      return (canvas.height - pad) - t * (canvas.height - pad*2);
    }

    function drawLine(arr, min, max, stroke){
      if (arr.length < 2) return;
      ctx.strokeStyle = stroke;
      ctx.lineWidth = 2;
      ctx.beginPath();
      for (let i=0; i<arr.length; i++){
        const x = (canvas.width/(MAX_POINTS-1))*i;
        const v = arr[i];
        const y = mapY(v, min, max);
        if (i === 0) ctx.moveTo(x,y);
        else ctx.lineTo(x,y);
      }
      ctx.stroke();
    }

    drawLine(seriesTemp, mmT.min, mmT.max, "rgba(232,221,198,.95)");
    drawLine(seriesHum,  mmH.min, mmH.max, "rgba(110,231,168,.85)");

    ctx.fillStyle = "rgba(241,239,231,.75)";
    ctx.font = "12px ui-sans-serif, system-ui";
    ctx.fillText("Temp (ivoire)", 12, 18);
    ctx.fillText("Hum (vert)", 140, 18);
  }

  const client = mqtt.connect(url, {
    username: USER,
    password: PASS,
    clean: true,
    reconnectPeriod: 1500,
    keepalive: 60,
  });

  client.on("connect", () => {
    setStatus(true, "connected", "MQTT OK");
    logLine("connected → subscribe telemetry/status");

    client.subscribe(TOPIC_TELEMETRY, { qos: 0 });
    client.subscribe(TOPIC_STATUS, { qos: 0 });
  });

  client.on("close", () => {
    setStatus(false, "disconnected", "retrying…");
    logLine("disconnected");
  });

  client.on("error", (err) => {
    setStatus(false, "error", "check creds/host");
    logLine("error: " + (err?.message || err));
  });

  function publishCmd(cmd){
    if (!client.connected){
      logLine("CMD blocked (not connected): " + cmd);
      return;
    }
    client.publish(TOPIC_CMD, cmd, { qos: 0, retain: false });
    logLine("cmd → " + TOPIC_CMD + " : " + cmd);
  }

  btnBeep.addEventListener("click", () => publishCmd("beep"));
  btnTheme.addEventListener("click", () => publishCmd("theme"));
  btnThemeOff.addEventListener("click", () => publishCmd("theme_off"));

  toggleBuzzer.addEventListener("change", () => {
    publishCmd(toggleBuzzer.checked ? "buzzer_on" : "buzzer_off");
  });

  btnClear.addEventListener("click", () => {
    logEl.textContent = "";
  });

  function setModeBadge(mode){
    const m = (mode || "").toLowerCase();
    elMode.textContent = (mode || "—");

    if (m === "dark") elMode.style.background = "rgba(232,221,198,.10)";
    else if (m === "hot") elMode.style.background = "rgba(255,107,107,.12)";
    else if (m === "humid") elMode.style.background = "rgba(110,231,168,.12)";
    else elMode.style.background = "rgba(232,221,198,.08)";
  }

  client.on("message", (topic, payload) => {
    const txt = payload.toString();

    if (topic === TOPIC_STATUS){
      elAlive.textContent = txt;
      logLine(`status ← ${txt}`);
      return;
    }

    if (topic !== TOPIC_TELEMETRY) return;

    try{
      const j = JSON.parse(txt);

      if (typeof j.temp_c === "number") elTemp.textContent = j.temp_c.toFixed(1);
      else elTemp.textContent = "—";

      if (typeof j.hum_pct === "number") elHum.textContent = j.hum_pct.toFixed(1);
      else elHum.textContent = "—";

      if (typeof j.lux_avg === "number") elLux.textContent = j.lux_avg;
      else if (typeof j.lux_raw === "number") elLux.textContent = j.lux_raw;
      else elLux.textContent = "—";

      setModeBadge(j.mode);

      if (typeof j.temp_c === "number") pushPoint(seriesTemp, j.temp_c);
      if (typeof j.hum_pct === "number") pushPoint(seriesHum, j.hum_pct);
      drawChart();

      const now = new Date();
      elLast.textContent = now.toLocaleTimeString("fr-CA");

      logLine(`telemetry ← temp=${j.temp_c} hum=${j.hum_pct} lux=${j.lux_avg ?? j.lux_raw} mode=${j.mode}`);
    }catch(e){
      logLine("telemetry parse error");
    }
  });

  drawChart();
  setStatus(false, "disconnected", "waiting…");
  elAlive.textContent = "—";
  setModeBadge("—");
</script>
